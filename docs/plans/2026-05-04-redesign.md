// Sparkline SVG
function Sparkline({ data, color }: { data: number[]; color: string }) {
  const w = 100, h = 28;
  const max = Math.max(...data), min = Math.min(...data), span = max - min || 1;
  const pts = data.map((d, i) => {
    const x = (i / (data.length - 1)) * w;
    const y = h - ((d - min) / span) * (h - 4) - 2;
    return `${x},${y}`;
  });
  const path = pts.map((p, i) => `${i === 0 ? 'M' : 'L'}${p}`).join(' ');
  const area = `${path} L${w},${h} L0,${h} Z`;
  return (
    <svg viewBox={`0 0 ${w} ${h}`} preserveAspectRatio="none" className="w-full" style={{ height: h }}>
      <path d={area} fill={color} opacity="0.12" />
      <path d={path} fill="none" stroke={color} strokeWidth="1.5" strokeLinejoin="round" strokeLinecap="round" />
    </svg>
  );
}

// Status pill
function Pill({ status }: { status: string }) {
  const map: Record<string, string> = {
    confirmed: 'pill-paid', pending: 'pill-pending', negotiating: 'pill-future',
    cleared: 'pill-cleared', bounced: 'pill-bounced', overdue: 'pill-overdue',
  };
  return (
    <span className={`pill ${map[status] ?? 'pill-future'}`}>
      <span className="dot" />{status}
    </span>
  );
}
